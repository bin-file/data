Aim: Implement gradient descent from scratch Code:
import numpy as np
x = np.array([1,2,3,4])
y = np.array([2,4,6,8]) w = 0
for i in range(100): p = w*x
w = w - 0.01*np.mean((p-y)*x)
print("Weight:", w) print("Prediction:", w*5)

Practical no 4
Aim: CNN from scratch Code:
import tensorflow as tf
(x,y),(xt,yt) = tf.keras.datasets.mnist.load_data() x = x[...,None] / 255
model = tf.keras.Sequential([ tf.keras.layers.Conv2D(8,3,activation="relu",
input_shape=(28,28,1)), tf.keras.layers.Flatten(), tf.keras.layers.Dense(10,activation="softmax")
])
model.compile("adam","sparse_categorical_crossentropy", metrics=["accuracy"])

model.fit(x,y,epochs=1) p = model.predict(x[:1])
print("Predicted:", p.argmax()) print("Actual:", y[0])

Practical no 6
Aim: Q-Learning Implementation Code:
import numpy as np


Q = np.zeros((5, 2))


for i in range(100): s = 0
for j in range(5):
a = np.argmax(Q[s]) ns = min(s + 1, 4)
r = 1 if ns == 4 else 0
Q[s,a] += 0.1 * (r + 0.9 * max(Q[ns]) - Q[s,a]) s = ns

print(Q)

import optuna


def objective(trial):
