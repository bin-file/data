Practical no 3
Aim: Edge detection (Sobel, Canny) Code:
import cv2

image = cv2.imread(r"C:\Users\Vivek wadher\Downloads\images.jpg") if image is None:
print("Image not found!") exit()

gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) average = cv2.blur(gray, (5,5))
gaussian = cv2.GaussianBlur(gray, (5,5), 0) median = cv2.medianBlur(gray, 5)
sobelx = cv2.Sobel(gray, cv2.CV_64F, 1, 0, ksize=3) sobely = cv2.Sobel(gray, cv2.CV_64F, 0, 1, ksize=3) sobel = cv2.magnitude(sobelx, sobely)
canny = cv2.Canny(gray, 100, 200) cv2.imshow("Original Image", gray) cv2.imshow("Average Filter", average) cv2.imshow("Gaussian Filter", gaussian) cv2.imshow("Median Filter", median)
cv2.imshow("Sobel Edge Detection", sobel / sobel.max()) cv2.imshow("Canny Edge Detection", canny)

cv2.waitKey(0) cv2.destroyAllWindows() 


Practical no 4
Aim: Corner detection (Harris) Code:
import cv2
import numpy as np
import matplotlib.pyplot as plt
image = cv2.imread(r"C:\Users\Vivek wadher\Downloads\images.jpg") if image is None:
print("Error: Image not found")
exit()

gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) gray = np.float32(gray)
harris_response = cv2.cornerHarris( gray,
blockSize=2, ksize=3, k=0.04
)

harris_response = cv2.dilate(harris_response, None) threshold = 0.01 * harris_response.max() image[harris_response > threshold] = [0, 0, 255]

plt.figure(figsize=(10,5))

plt.subplot(1,2,1) plt.title("Original Image")
plt.imshow(cv2.cvtColor(cv2.imread(r"C:\Users\Vivek wadher\Downloads\images.jpg"), cv2.COLOR_BGR2RGB)) plt.axis("off")
plt.subplot(1,2,2)
plt.title("Harris Corner Detection") plt.imshow(cv2.cvtColor(image, cv2.COLOR_BGR2RGB)) plt.axis("off")
plt.show()


Practical no 5
Aim: Feature extraction using SIFT/ORB Code:
import cv2
import matplotlib.pyplot as plt

img1 = cv2.imread(r"C:\Users\Vivek wadher\Downloads\images.jpg") img2 = cv2.imread(r"C:\Users\Vivek wadher\Downloads\images (1).jpg")

if img1 is None or img2 is None: print("Error: Image not found.") exit()

gray1 = cv2.cvtColor(img1, cv2.COLOR_BGR2GRAY) gray2 = cv2.cvtColor(img2, cv2.COLOR_BGR2GRAY)

sift = cv2.SIFT_create()

kp1, des1 = sift.detectAndCompute(gray1, None) kp2, des2 = sift.detectAndCompute(gray2, None)

bf = cv2.BFMatcher(cv2.NORM_L2, crossCheck=True) matches = bf.match(des1, des2)
matches = sorted(matches, key=lambda x: x.distance)

result = cv2.drawMatches( img1, kp1,
img2, kp2, matches[:30], None,
flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS
)

plt.figure(figsize=(15,6)) plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB)) plt.title("SIFT Feature Matching")
plt.axis("off") plt.show()



Practical no 8
Aim: Object Detection using (YOLO)
Code:
from ultralytics import YOLO import cv2

model = YOLO("yolov8n.pt")

cap = cv2.VideoCapture(0) while True:
ret, frame = cap.read()

if not ret:
break

results = model(frame) annotated_frame = results[0].plot()
cv2.imshow("Live Object Detection", annotated_frame)

if cv2.waitKey(1) & 0xFF == ord('q'): break

cap.release() cv2.destroyAllWindows()
